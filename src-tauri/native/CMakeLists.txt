cmake_minimum_required(VERSION 3.20)
project(VidExtractTranslate LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WITH_MKL OFF CACHE BOOL "" FORCE)
set(WITH_CUDA OFF CACHE BOOL "" FORCE)
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(WITH_CPU_DISPATCH ON CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${CT2_ROOT}" ctranslate2)
add_subdirectory("${SPM_ROOT}" sentencepiece)
add_executable(translate-cpu translate.cpp)
target_include_directories(translate-cpu PRIVATE "${CT2_ROOT}/include" "${SPM_ROOT}/src" "${JSON_ROOT}")
target_link_libraries(translate-cpu PRIVATE ctranslate2 sentencepiece-static)
if(MSVC)
  target_sources(translate-cpu PRIVATE translate.manifest)
endif()
